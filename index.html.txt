        // Load from localStorage if present
        function loadState() {
            try {
                const saved = localStorage.getItem('pulse_matrix_state_v2');
                if (saved) {
                    const parsed = JSON.parse(saved);
                    appState = { ...appState, ...parsed };
                }
            } catch (e) {
                console.error("Storage load error:", e);
            }
        }

        // Save to localStorage immediately on any change
        function saveState() {
            try {
                localStorage.setItem('pulse_matrix_state_v2', JSON.stringify(appState));
                showSyncBadge();
            } catch (e) {
                console.error("Storage save error:", e);
            }
        }

            setTimeout(() => {
                badge.innerHTML = '<i class="fa-solid fa-cloud-arrow-up text-[#B87D59] mr-1"></i> Saved Locally';
            }, 1200);
        }

        let currentFilterMuscle = 'All';

        function renderExerciseLibrary(searchQuery = '') {
            const container = document.getElementById('exercise-library-list');
            container.innerHTML = '';
            
            const filtered = EXERCISE_LIBRARY.filter(ex => {
                const matchMuscle = currentFilterMuscle === 'All' || ex.muscle === currentFilterMuscle;
                const matchSearch = ex.name.toLowerCase().includes(searchQuery.toLowerCase()) || ex.muscle.toLowerCase().includes(searchQuery.toLowerCase());
                return matchMuscle && matchSearch;
            });

            document.getElementById('library-count').innerText = `${filtered.length} items`;

            if (filtered.length === 0) {
                container.innerHTML = `<div class="text-xs text-center py-4 text-[#7A6A5B]">No matching exercises found.</div>`;
                return;
            }

            filtered.forEach(ex => {
                const div = document.createElement('div');
                div.className = "flex items-center justify-between p-2.5 bg-white border border-[#E2D8CC] rounded-xl hover:border-[#B87D59] transition cursor-pointer";
                div.innerHTML = `
                    <div>
                        <div class="text-xs font-bold text-[#4A3B32]">${ex.name}</div>
                        <div class="text-[10px] text-[#7A6A5B]">${ex.muscle}</div>
                    </div>
                    <button onclick="addExerciseToFirstRoutine('${ex.name}', '${ex.muscle}')" class="w-7 h-7 bg-[#EFEAE1] hover:bg-[#B87D59] hover:text-white rounded-lg text-xs font-bold flex items-center justify-center transition">
                        <i class="fa-solid fa-plus"></i>
                    </button>
                `;
                container.appendChild(div);
            });
        }

        function filterExerciseLibrary() {

        function renderRoutines() {
            const container = document.getElementById('routine-days-container');
            container.innerHTML = '';

            appState.routines.forEach((routine, rIdx) => {
                const card = document.createElement('div');
                card.className = "card-warm p-5 rounded-2xl space-y-4";
                
                let exercisesHTML = '';
                routine.exercises.forEach((ex, eIdx) => {
                    // Calculate working max based on highest weight set
                    let maxWeight = 0;
                    ex.sets.forEach(s => { if (s.weight > maxWeight) maxWeight = s.weight; });

                    let setsHTML = '';
                    ex.sets.forEach((set, sIdx) => {
                        setsHTML += `
                            <div class="flex items-center space-x-2 bg-white p-2 rounded-xl border border-[#E2D8CC]">
                                <span class="text-xs font-bold text-[#7A6A5B] w-12">Set ${sIdx+1}</span>
                                <div class="flex-1 flex items-center space-x-1">
                                    <input type="number" value="${set.weight}" onchange="updateSetWeight(${rIdx}, ${eIdx}, ${sIdx}, this.value)" class="w-16 px-2 py-1 bg-[#F7F4EF] rounded-lg text-xs font-bold text-center border border-[#E2D8CC]">
                                    <span class="text-xs text-[#7A6A5B]">kg</span>
                                </div>
                                <div class="flex-1">
                                    <select onchange="updateSetReps(${rIdx}, ${eIdx}, ${sIdx}, this.value)" class="w-full px-2 py-1 bg-[#F7F4EF] rounded-lg text-xs font-bold border border-[#E2D8CC]">
                                        ${generateRepOptions(set.reps)}
                                    </select>
                                </div>
                            </div>
                        `;
                    });

                    exercisesHTML += `
                        <div class="bg-[#FAF7F2] p-4 rounded-xl border border-[#E2D8CC] space-y-3">
                            <div class="flex items-center justify-between">
                                <div>
                                    <h4 class="font-bold text-sm text-[#4A3B32]">${ex.name}</h4>
                                    <div class="flex items-center space-x-2 mt-0.5">
                                        <span class="text-[10px] bg-[#EFEAE1] px-2 py-0.5 rounded text-[#7A6A5B]">${ex.muscle}</span>
                                        <span class="text-[10px] text-[#B87D59] font-medium">Target: ${ex.targetRange} reps</span>
                                        <span class="text-[10px] text-[#4A3B32] font-bold">Max: ${maxWeight}kg</span>
                                    </div>
                                </div>
                                <button onclick="removeExerciseFromRoutine(${rIdx}, ${eIdx})" class="text-[#A89A8B] hover:text-red-600 text-xs"><i class="fa-solid fa-trash"></i></button>
                            </div>
                            <div class="grid grid-cols-1 md:grid-cols-2 gap-2">
                                ${setsHTML}
                            </div>
                        </div>
                    `;
                });

                card.innerHTML = `
                    <div class="flex items-center justify-between pb-2 border-b border-[#E2D8CC]">
                        <div>
                            <h3 class="font-bold text-base text-[#4A3B32]">${routine.title}</h3>
                            <p class="text-xs text-[#7A6A5B]">${routine.exercises.length} exercises configured</p>
                        </div>
                        <button onclick="removeRoutineDay(${rIdx})" class="text-xs text-[#A89A8B] hover:text-red-600"><i class="fa-solid fa-trash mr-1"></i>Delete Routine</button>
                    </div>
                    <div class="space-y-3">
                        ${exercisesHTML || '<div class="text-xs text-center py-4 text-[#7A6A5B]">No exercises in this routine yet. Click + on any exercise from the library.</div>'}
                    </div>
                `;
                container.appendChild(card);
            });
        }

        function generateRepOptions(selectedReps) {
            let opts = '';
            for (let i = 0; i <= 30; i++) {
                const sel = i === Number(selectedReps) ? 'selected' : '';
                opts += `<option value="${i}" ${sel}>${i} reps</option>`;
            }
            return opts;
        }

        function updateSetWeight(rIdx, eIdx, sIdx, val) {
            appState.routines[rIdx].exercises[eIdx].sets[sIdx].weight = Number(val);
            saveState();
            renderRoutines();
        }

        function updateSetReps(rIdx, eIdx, sIdx, val) {
            appState.routines[rIdx].exercises[eIdx].sets[sIdx].reps = Number(val);
            saveState();
            renderRoutines();
        }

        function addExerciseToFirstRoutine(name, muscle) {
            if (appState.routines.length === 0) {
                addNewRoutineDay();
            }
            appState.routines[0].exercises.push({
                id: 'ex_' + Date.now(),
                name: name,
                muscle: muscle,
                targetRange: '8-10',
                sets: [{ weight: 20, reps: 10 }, { weight: 20, reps: 10 }, { weight: 20, reps: 10 }]
            });
            saveState();
            renderRoutines();
        }

        function removeExerciseFromRoutine(rIdx, eIdx) {
            appState.routines[rIdx].exercises.splice(eIdx, 1);
            saveState();
            renderRoutines();
        }

        function addNewRoutineDay() {
            const title = prompt("Enter Routine Name (e.g. Upper Body Power):", "Full Body Day");

                id: 'routine_' + Date.now(),
                title: title,

            });

                saveState();
                renderRoutines();
            }
        }

        function getTodayNutrition() {
            const dateStr = appState.selectedNutritionDate;
            if (!appState.nutritionDays[dateStr]) {
                appState.nutritionDays[dateStr] = {
                    meals: [
                        { name: 'Singer Wrap + Tea', cal: 380, pro: 22, carb: 45, fat: 12, fibre: 4, sugar: 6, sodium: 520, potassium: 310 },
                        { name: 'Air-fried Chicken Breast', cal: 330, pro: 62, carb: 0, fat: 5, fibre: 0, sugar: 0, sodium: 410, potassium: 450 }
                    ],
                    water: 1.5
                };
            }
            return appState.nutritionDays[dateStr];
        }

        function renderNutrition() {
            document.getElementById('nutrition-date-picker').value = appState.selectedNutritionDate;
            document.getElementById('current-display-date').innerText = appState.selectedNutritionDate === new Date().toISOString().split('T')[0] ? 'Today' : appState.selectedNutritionDate;

            const dayData = getTodayNutrition();
            let totalCal = 0, totalPro = 0, totalCarb = 0, totalFat = 0, totalFibre = 0, totalSugar = 0, totalSodium = 0, totalPotassium = 0;

            dayData.meals.forEach(m => {
                totalCal += Number(m.cal || 0);
                totalPro += Number(m.pro || 0);
                totalCarb += Number(m.carb || 0);
                totalFat += Number(m.fat || 0);
                totalFibre += Number(m.fibre || 0);
                totalSugar += Number(m.sugar || 0);
                totalSodium += Number(m.sodium || 0);
                totalPotassium += Number(m.potassium || 0);
            });

            // Update Progress Bars (Targets: 2000 kcal, 180g pro, 200g carb, 65g fat)
            document.getElementById('cal-ratio').innerText = `${totalCal} / 2000 kcal`;
            document.getElementById('bar-cal').style.width = `${Math.min(100, (totalCal / 2000) * 100)}%`;

            document.getElementById('pro-ratio').innerText = `${totalPro} / 180g`;
            document.getElementById('bar-pro').style.width = `${Math.min(100, (totalPro / 180) * 100)}%`;

            document.getElementById('carb-ratio').innerText = `${totalCarb} / 200g`;
            document.getElementById('bar-carb').style.width = `${Math.min(100, (totalCarb / 200) * 100)}%`;

            document.getElementById('fat-ratio').innerText = `${totalFat} / 65g`;
            document.getElementById('bar-fat').style.width = `${Math.min(100, (totalFat / 65) * 100)}%`;

            // Micro stats
            document.getElementById('stat-fibre').innerText = `${totalFibre}g / 30g`;
            document.getElementById('stat-sugar').innerText = `${totalSugar}g / 40g`;
            document.getElementById('stat-sodium').innerText = `${totalSodium}mg / 2300mg`;
            document.getElementById('stat-potassium').innerText = `${totalPotassium}mg / 3500mg`;
            document.getElementById('stat-water').innerText = `${dayData.water.toFixed(1)}L / 3.0L`;

            // Render meals list
            const container = document.getElementById('meals-list-container');
            container.innerHTML = '';
            document.getElementById('meal-count-badge').innerText = `${dayData.meals.length} meals`;

            if (dayData.meals.length === 0) {
                container.innerHTML = `<div class="text-xs text-center py-6 text-[#7A6A5B]">No meals logged for this date. Snap a photo or add manually.</div>`;
                return;
            }

            dayData.meals.forEach((m, idx) => {
                const div = document.createElement('div');
                div.className = "bg-white p-4 rounded-xl border border-[#E2D8CC] flex items-center justify-between";
                div.innerHTML = `
                    <div>
                        <h4 class="font-bold text-sm text-[#4A3B32]">${m.name}</h4>
                        <div class="flex flex-wrap gap-2 mt-1 text-xs text-[#7A6A5B]">
                            <span class="bg-[#F7F4EF] px-2 py-0.5 rounded font-medium">${m.cal} kcal</span>
                            <span class="bg-[#F7F4EF] px-2 py-0.5 rounded font-medium">P: ${m.pro}g</span>
                            <span class="bg-[#F7F4EF] px-2 py-0.5 rounded font-medium">C: ${m.carb}g</span>
                            <span class="bg-[#F7F4EF] px-2 py-0.5 rounded font-medium">F: ${m.fat}g</span>

                container.appendChild(div);
